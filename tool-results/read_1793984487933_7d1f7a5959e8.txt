   224→function Overview({ data, hub }: { data: DetailData; hub: string }) {
   225→  const { media } = data;
   226→  const extra = media.extra as {
   227→    engine?: string; platforms?: string[]; sizeGb?: number; version?: string;
   228→    officialUrl?: string; requirements?: Record<string, string>;
   229→    permissions?: { domain: string; reason: string }[];
   230→    installs?: number; sourceUrl?: string; kind?: string;
   231→  };
   232→  return (
   233→    <div className="grid gap-6 lg:grid-cols-[1.6fr_1fr]">
   234→      <div className="space-y-5">
   235→        <div>
   236→          <h3 className="mb-2 text-[15px] font-bold">Synopsis</h3>
   237→          <p className="text-[14px] leading-relaxed text-foreground/85">{media.description}</p>
   238→        </div>
   239→
   240→        {/* category-specific blocks */}
   241→        {media.category === "GAME" && (
   242→          <div className="rounded-xl border border-border bg-card p-4">
   243→            <h3 className="flex items-center gap-2 text-[15px] font-bold"><Gamepad2 className="h-4 w-4" style={{ color: hub }} /> Technical details</h3>
   244→            <dl className="mt-3 grid grid-cols-2 gap-x-6 gap-y-2 text-[13px] sm:grid-cols-3">
   245→              {extra.engine && <Meta label="Engine" value={extra.engine} />}
   246→              {extra.version && <Meta label="Version" value={extra.version} />}
   247→              {extra.sizeGb && <Meta label="Size" value={`${extra.sizeGb} GB`} />}
   248→              {extra.platforms && <Meta label="Platforms" value={extra.platforms.join(", ")} />}
   249→              {extra.requirements && Object.entries(extra.requirements).map(([k, v]) => (
   250→                <Meta key={k} label={k} value={v} />
   251→              ))}
   252→            </dl>
   253→          </div>
   254→        )}
   255→
   256→        {media.category === "SCRIPT" && extra.permissions && (
   257→          <div className="rounded-xl border border-border bg-card p-4">
   258→            <h3 className="flex items-center gap-2 text-[15px] font-bold"><FileCode2 className="h-4 w-4" style={{ color: hub }} /> Requested permissions</h3>
   259→            <p className="mt-1 text-xs text-muted-foreground">
   260→              This platform never executes third-party code. Installs are tracked metadata with enable/disable control (sandboxed).
   261→            </p>
   262→            <ul className="mt-3 space-y-1.5">
   263→              {extra.permissions.map((p) => (
   264→                <li key={p.domain} className="flex items-center gap-2 text-[13px]">
   265→                  <Badge variant="secondary" className="rounded-md bg-background font-mono text-[11px]">{p.domain}</Badge>
   266→                  <span className="text-muted-foreground">{p.reason}</span>
   267→                </li>
   268→              ))}
   269→            </ul>
   270→          </div>
   271→        )}
   272→
   273→        {media.tags.length > 0 && (
   274→          <div>
   275→            <h3 className="mb-2 text-[15px] font-bold">Tags</h3>
   276→            <div className="flex flex-wrap gap-1.5">
   277→              {media.tags.map((t) => (
   278→                <Badge key={t} variant="secondary" className="rounded-full bg-muted text-muted-foreground">{t}</Badge>
   279→              ))}
   280→            </div>
   281→          </div>
   282→        )}
   283→      </div>
   284→
   285→      <div className="space-y-4">
   286→        <div className="rounded-xl border border-border bg-card p-4">
   287→          <h3 className="mb-3 text-[13px] font-bold uppercase tracking-wide text-muted-foreground">Details</h3>
   288→          <dl className="space-y-2.5 text-[13px]">
   289→            <Meta label="Category" value={categoryMeta(media.category).label} />
   290→            {media.format && <Meta label="Format" value={media.format} />}
   291→            {media.status && <Meta label="Status" value={STATUS_LABELS[media.status] ?? media.status} />}
   292→            {media.year && <Meta label="Year" value={String(media.year)} />}
   293→            {media.studio && <Meta label="Studio" value={media.studio} />}
   294→            {media.creator && <Meta label={media.category === "GAME" ? "Developer" : "Author"} value={media.creator} />}
   295→            <Meta label="Age rating" value={media.ageRating} />
   296→            {media.altTitles.length > 0 && <Meta label="Alt titles" value={media.altTitles.join(", ")} />}
   297→          </dl>
   298→        </div>
   299→        {media.externalLinks.length > 0 && (
   300→          <div className="rounded-xl border border-border bg-card p-4">
   301→            <h3 className="mb-3 text-[13px] font-bold uppercase tracking-wide text-muted-foreground">Links</h3>
   302→            <ul className="space-y-2">
   303→              {media.externalLinks.map((l) => (
   304→                <li key={l.url}>
   305→                  <a href={l.url} target="_blank" rel="noreferrer" className="flex items-center gap-2 text-[13px] font-semibold hover:underline" style={{ color: hub }}>
   306→                    <ExternalLink className="h-3.5 w-3.5" /> {l.label}
   307→                  </a>
   308→                </li>
   309→              ))}
   310→            </ul>
   311→          </div>
   312→        )}
   313→        {data.people.length > 0 && (
   314→          <div className="rounded-xl border border-border bg-card p-4">
   315→            <h3 className="mb-3 text-[13px] font-bold uppercase tracking-wide text-muted-foreground">Credits</h3>
   316→            <ul className="space-y-2 text-[13px]">
   317→              {data.people.map((p) => (
   318→                <li key={p.id} className="flex items-center justify-between gap-2">
   319→                  <span className="font-semibold">{p.name}</span>
   320→                  <span className="text-muted-foreground">{p.role}</span>
   321→                </li>
   322→              ))}
   323→            </ul>
   324→          </div>
   325→        )}
   326→      </div>
   327→    </div>
   328→  );
   329→}
